// MARK: - Fonksiyonlar

func startSearch():
  isSearching = true
  statusMessage = "Rakip aranıyor..."
  Task {
    do {
      // 1. Kuyruğa katıl
      try await NetworkService.shared.joinQueue(userId: userId)
      // 2. Maç bul (polling — 10 kez 2 saniye aralıkla)
      for _ in 0..<10 {
        try await Task.sleep(nanoseconds: 2_000_000_000)
        if let mid = try? await NetworkService.shared.findMatch(userId: userId) {
          await MainActor.run {
            matchId = mid
            navigateToDuel = true
            isSearching = false
          }
          return
        }
      }
      // Maç bulunamadı
      await MainActor.run {
        isSearching = false
        statusMessage = "Rakip bulunamadı. Tekrar dene."
      }
    } catch {
      await MainActor.run {
        isSearching = false
        statusMessage = "Hata: \(error.localizedDescription)"
      }
    }
  }

=== 3. NetworkService.swift — şu fonksiyonları ekle ===

// Matchmaking
func joinQueue(userId: String) async throws:
  POST /api/Matchmaking/join-queue
  body: { "userId": userId }

func findMatch(userId: String) async throws -> String?:
  GET /api/Matchmaking/find-match/{userId}
  response: { "matchId": "..." } veya 404
  404 gelirse nil döndür, hata fırlatma

func finishMatch(matchId: String) async throws:
  POST /api/Matchmaking/finish-match/{matchId}
  body boş
