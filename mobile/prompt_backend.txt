=== 1. NetworkService.swift — komple yaz ===

import Foundation

class NetworkService {
    static let shared = NetworkService()
    private let baseURL = "http://localhost:5150"
    
    // MARK: - User
    func getUser(userId: String) async throws -> UserDTO
    
    // MARK: - Progress  
    func getWeeklyProgress(userId: String) async throws -> WeeklyProgressDTO
    
    // MARK: - Leaderboard
    func getLeaderboard(count: Int) async throws -> [LeaderboardDTO]
    
    // MARK: - Preferences
    func getPreferences(userId: String) async throws -> PreferencesDTO
    func updatePreferences(prefId: String, payload: UpdatePreferencesDTO) async throws -> PreferencesDTO
    
    // MARK: - Nickname
    func generateNickname(userId: String, keywords: [String]) async throws -> NicknameResponseDTO
    
    // MARK: - Generic request
    private func request<T: Decodable>(_ path: String, method: String = "GET", body: Encodable? = nil) async throws -> T
}

DTO'lar (backend response'larına uygun):

struct UserDTO: Codable {
    let id: String
    let username: String
    let level: Int
    let xp: Int
    let xpForLevel: Int
    let isPremium: Bool
}

struct WeeklyProgressDTO: Codable {
    let weeklyQuestionTarget: Int
    let totalAnswered: Int
    let percentage: Double
    let dailyBreakdown: [DailyBreakdownDTO]?
}

struct DailyBreakdownDTO: Codable {
    let day: String
    let count: Int
}

struct LeaderboardDTO: Codable {
    let rank: Int
    let username: String
    let level: Int
    let totalXP: Int
}

struct PreferencesDTO: Codable {
    let id: String
    let userId: String
    let focusAreas: [String]
    let weeklyQuestionTarget: Int
    let dailyTarget: Int
    let goalSubject: String?
}

struct UpdatePreferencesDTO: Codable {
    let focusAreas: [String]
    let weeklyQuestionTarget: Int
    let dailyTarget: Int
    let goalSubject: String?
}

struct NicknameResponseDTO: Codable {
    let nicknames: [String]?
    let nickname: String?
}

Hata yönetimi:
enum APIError: Error {
    case invalidURL
    case noData
    case decodingError(Error)
    case httpError(Int)
    case networkError(Error)
}

func savePreferences(userId: String) async:
  guard let prefId = prefId else { return }
  let payload = UpdatePreferencesDTO(

=== 4. EducaiApp.swift — Login flow ekle ===

RootView'a şunları ekle:
@State private var isLoggedIn: Bool = false
@State private var userId: String = ""

// App açılınca UserDefaults'tan kontrol et:
.onAppear {
  if let saved = UserDefaults.standard.string(forKey: "userId") {
    userId = saved
    isLoggedIn = true
    Task { await model.loadFromBackend(userId: saved) }
  }
}

isLoggedIn false ise: LoginView göster
isLoggedIn true ise: mevcut tab bar + ProfileView/PreferencesView göster

LoginView'dan onLogin callback:
onLogin: { uid in
  userId = uid
  isLoggedIn = true
  Task { await model.loadFromBackend(userId: uid) }
}

=== KURALLAR ===
- iOS 16+ uyumlu
- async/await kullan, completion handler yok
- Tüm network çağrıları MainActor'da UI güncelle:
  await MainActor.run { self.username = ... }
- try/catch her yerde
- Mock veri fallback olarak korunsun
